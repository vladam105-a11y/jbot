import asyncio
import logging
import os

from aiogram import Bot, Dispatcher, F
from aiogram.filters import CommandStart
from aiogram.types import Message, ReplyKeyboardMarkup, KeyboardButton
from aiogram.client.default import DefaultBotProperties
from aiogram.enums import ParseMode


# =========================
# НАСТРОЙКИ
# =========================

TOKEN = os.getenv("BOT_TOKEN")

if not TOKEN:
    raise ValueError("Переменная окружения BOT_TOKEN не установлена")


# =========================
# СОЗДАНИЕ БОТА
# =========================

bot = Bot(
    token=TOKEN,
    default=DefaultBotProperties(
        parse_mode=ParseMode.HTML
    )
)

dp = Dispatcher()


# =========================
# КЛАВИАТУРА
# =========================

main_keyboard = ReplyKeyboardMarkup(
    keyboard=[
        [
            KeyboardButton(text="О нас"),
            KeyboardButton(text="Направления"),
        ],
        [
            KeyboardButton(text="Контакты"),
        ],
    ],
    resize_keyboard=True,
    input_field_placeholder="Выберите раздел"
)


# =========================
# /START
# =========================

@dp.message(CommandStart())
async def start_handler(message: Message):
    await message.answer(
        "<b>Молодёжный ресурсный центр "
        "Есильского района СКО</b> 👋\n\n"
        "Твоя энергия. Твои идеи. Твои возможности.\n\n"
        "Мы — пространство для поддержки, развития и объединения молодёжи. "
        "Помогаем реализовывать инициативы, участвовать в общественной жизни, "
        "волонтёрстве, развиваться и находить возможности для себя.\n\n"
        "Выберите интересующий раздел ниже:",
        reply_markup=main_keyboard

# =========================
# ЗАПУСК
# =========================

async def main():
    logging.basicConfig(level=logging.INFO)

    # Удаляем старые необработанные обновления
    await bot.delete_webhook(drop_pending_updates=True)

    print("Бот МРЦ запущен!")

    await dp.start_polling(bot)


if __name__ == "__main__":
    asyncio.run(main())
